package es.maetd.bff.external.web.domain.model;

import es.maetd.marketcreators.ddd.core.model.DomainModel;
import javax.validation.constraints.NotEmpty;
import lombok.*;

@Getter
@ToString
@NoArgsConstructor(access = AccessLevel.PRIVATE)
@AllArgsConstructor(access = AccessLevel.PRIVATE)
public class ExampleModel extends DomainModel {

    private String id;

    @NotEmpty
    private String requiredProp;

    public static ExampleModel createDomainObject(String requiredProp) {

        ExampleModel domainObj = new ExampleModel(null, requiredProp);

        domainObj.validate();

        return domainObj;
    }
}
